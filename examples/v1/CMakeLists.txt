cmake_minimum_required(VERSION 3.26)
project(h5pp-v1-examples CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SetupExecutableDirectory.cmake")
h5pp_setup_executable_directory()

file(GLOB EXAMPLES CONFIGURE_DEPENDS "*.cpp")
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/output)
add_custom_target(h5pp-v1-example-all)
foreach(example ${EXAMPLES})
    get_filename_component(example_name ${example} NAME_WE)
    add_executable(h5pp-v1-${example_name} ${example})
    target_link_libraries(h5pp-v1-${example_name} PRIVATE h5pp)
    target_compile_definitions(h5pp-v1-${example_name} PRIVATE H5PP_EXAMPLE_DIR=\"output/\")
    if(MSVC)
        target_link_options(h5pp-v1-${example_name} PRIVATE /ignore:4099)
    endif()
    add_dependencies(h5pp-v1-example-all h5pp-v1-${example_name})
endforeach()
