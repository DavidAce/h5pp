cmake_minimum_required(VERSION 3.26)
project(h5pp-v1-tests CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SetupExecutableDirectory.cmake")
h5pp_setup_executable_directory()

find_package(Catch2 3 REQUIRED)

file(GLOB TESTS CONFIGURE_DEPENDS "*.cpp")
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/output)
add_custom_target(h5pp-v1-test-all)
foreach(test ${TESTS})
    get_filename_component(test_name ${test} NAME_WE)
    add_executable(h5pp-v1-${test_name} ${test})
    target_link_libraries(h5pp-v1-${test_name} PRIVATE h5pp Catch2::Catch2)
    # Use Catch2's __LINE__ fallback to avoid Clang 22 warnings about __COUNTER__.
    target_compile_definitions(h5pp-v1-${test_name} PRIVATE H5PP_TEST_DIR=\"output/\" CATCH_CONFIG_NO_COUNTER)
    add_test(NAME h5pp-v1-${test_name} WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} COMMAND h5pp-v1-${test_name})
    if(MSVC)
        target_compile_definitions(h5pp-v1-${test_name} PRIVATE _SILENCE_NONFLOATING_COMPLEX_DEPRECATION_WARNING)
        target_link_options(h5pp-v1-${test_name} PRIVATE /ignore:4099)
    endif()
    if(H5PP_ENABLE_COVERAGE)
        target_compile_options(h5pp-v1-${test_name} PRIVATE --coverage)
        target_link_options(h5pp-v1-${test_name} PRIVATE --coverage)
    endif()
    add_dependencies(h5pp-v1-test-all h5pp-v1-${test_name})
endforeach()
